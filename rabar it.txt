      /* rebuild with same punctuation style */
      return word.replace(clean, found);
    }
  }
  return null;
}

function translateText(text, kuToEn){
  if (!text.trim()) return '';
  /* keep whitespace and newlines */
  const parts = text.split(/(\s+)/);
  return parts.map(part => {
    if (/^\s+$/.test(part)) return part;
    const tr = translateWord(part, kuToEn);
    return tr !== null ? tr : part;
  }).join('');
}

  if (isKuToEn) {
    badgeFrom.textContent = 'کوردی';  badgeFrom.classList.add('active');
    badgeTo.textContent   = 'English'; badgeTo.classList.remove('active');
    labelFrom.textContent = 'کوردی (Kurdish)';
    labelTo.textContent   = 'English (ئینگلیزی)';
    inputText.placeholder  = 'لێرە دەقەکەت بنووسە...';
    outputText.placeholder = 'وەرگێڕان لێرە دەردەکەوێت...';
    inputText.dir = 'rtl';
  } else {
    badgeFrom.textContent = 'English'; badgeFrom.classList.add('active');
    badgeTo.textContent   = 'کوردی';   badgeTo.classList.remove('active');
    labelFrom.textContent = 'English (ئینگلیزی)';
    labelTo.textContent   = 'کوردی (Kurdish)';
    inputText.placeholder  = 'Type your text here...';
    outputText.placeholder = 'Translation appears here...';
    inputText.dir = 'ltr';
  }
}
